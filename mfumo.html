<script>
// Default Event Data
let eventConfig = {
title: "Tamasha Kuu la Musiki & Utamaduni",
date: "15 Desemba 2026",
time: "Saa 12:00 Jioni - Saa 6:00 Usiku",
venue: "Uwanja wa Mpira, Dodoma",
price: 15000,
image: "https://images.unsplash.com/photo-1514525253161-7a46d19cd819?auto=format&fit=crop&w=800&q=80"
};

// Ticket Database (Simulated in Local Storage)
let ticketDatabase = JSON.parse(localStorage.getItem('event_tickets')) || [];

let html5QrcodeScanner = null;

// Initialize icons and display
window.onload = function() {
lucide.createIcons();
loadEventDisplay();
};

// Tab Switcher
function switchTab(tabId) {
document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));
document.getElementById(tabId).classList.add('active');

if (tabId === 'scanner-page') {
startScanner();
} else {
stopScanner();
}
}

// Load Event Data to Public View
function loadEventDisplay() {
document.getElementById('display-title').innerText = eventConfig.title;
document.getElementById('display-date').innerText = eventConfig.date;
document.getElementById('display-time').innerText = eventConfig.time;
document.getElementById('display-venue').innerText = eventConfig.venue;
document.getElementById('display-price-tag').innerText = `Tsh ${Number(eventConfig.price).toLocaleString()} Kamili`;
document.getElementById('btn-price').innerText = Number(eventConfig.price).toLocaleString();
document.getElementById('display-image').src = eventConfig.image;

// Fill Admin Form defaults
document.getElementById('cfg-title').value = eventConfig.title;
document.getElementById('cfg-date').value = eventConfig.date;
document.getElementById('cfg-time').value = eventConfig.time;
document.getElementById('cfg-venue').value = eventConfig.venue;
document.getElementById('cfg-price').value = eventConfig.price;
document.getElementById('cfg-image').value = eventConfig.image;
}

// Save Admin Configuration
function saveEventConfig(e) {
e.preventDefault();
eventConfig = {
title: document.getElementById('cfg-title').value,
date: document.getElementById('cfg-date').value,
time: document.getElementById('cfg-time').value,
venue: document.getElementById('cfg-venue').value,
price: document.getElementById('cfg-price').value,
image: document.getElementById('cfg-image').value
};
loadEventDisplay();
alert('Taarifa za tukio zimehifadhiwa vyema!');
switchTab('event-page');
}

// Handle Purchase Simulation
function handlePurchase(e) {
e.preventDefault();
const name = document.getElementById('buyer-name').value;
const phone = document.getElementById('buyer-phone').value;
const ticketId = 'TCK-' + Math.floor(100000 + Math.random() * 900000);

const newTicket = {
id: ticketId,
name: name,
phone: phone,
amount: `Tsh ${Number(eventConfig.price).toLocaleString()}`,
eventTitle: eventConfig.title,
checkedIn: false
};

ticketDatabase.push(newTicket);
localStorage.setItem('event_tickets', JSON.stringify(ticketDatabase));

// Render Ticket UI
document.getElementById('ticket-event-name').innerText = newTicket.eventTitle;
document.getElementById('ticket-id-display').innerText = `ID ya Tiketi: #${newTicket.id}`;
document.getElementById('ticket-holder').innerText = newTicket.name;
document.getElementById('ticket-phone').innerText = newTicket.phone;
document.getElementById('ticket-amount').innerText = newTicket.amount;

// Generate QR Code
const qrContainer = document.getElementById('qrcode-container');
qrContainer.innerHTML = '';
new QRCode(qrContainer, {
text: newTicket.id,
width: 160,
height: 160
});

document.getElementById('generated-ticket').classList.remove('hidden');
document.getElementById('purchase-form').reset();
alert(`Malipo ya Tsh ${newTicket.amount} yamefanikiwa! Tiketi yako ipo tayari chini.`);
}

// Scanner Logic
function startScanner() {
if (!html5QrcodeScanner) {
html5QrcodeScanner = new Html5QrcodeScanner("qr-reader", { fps: 10, qrbox: 250 });
}
html5QrcodeScanner.render(onScanSuccess);
}

function stopScanner() {
if (html5QrcodeScanner) {
html5QrcodeScanner.clear().catch(error => console.error("Scanner clear error", error));
}
}

function onScanSuccess(decodedText) {
const resultDiv = document.getElementById('scan-result');
const titleEl = document.getElementById('scan-status-title');
const descEl = document.getElementById('scan-status-desc');

resultDiv.classList.remove('hidden', 'bg-green-100', 'bg-red-100', 'text-green-900', 'text-red-900');

// Find ticket in local database
const foundTicket = ticketDatabase.find(t => t.id === decodedText);

if (!foundTicket) {
resultDiv.classList.add('bg-red-100', 'text-red-900');
titleEl.innerText = "❌ Tiketi Haipo / Si Halali!";
descEl.innerText = `Namba ya tiketi (${decodedText}) haionekani kwenye mfumo.`;
} else if (foundTicket.checkedIn) {
resultDiv.classList.add('bg-red-100', 'text-red-900');
titleEl.innerText = "⚠️ TIKETI HII IMESHATUMIKA!";
descEl.innerText = `Mwenye Tiketi: ${foundTicket.name} (Tayari aliingia mlangoni).`;
} else {
foundTicket.checkedIn = true;
localStorage.setItem('event_tickets', JSON.stringify(ticketDatabase));

resultDiv.classList.add('bg-green-100', 'text-green-900');
titleEl.innerText = "✅ TIKETI HALALI - RUHUSU KUINGIA";
descEl.innerText = `Mwenye Tiketi: ${foundTicket.name} | Simu: ${foundTicket.phone}`;
}
}
